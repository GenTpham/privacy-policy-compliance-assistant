    """
    backend/app/api/sources.py
    GET /api/sources — returns sorted list of distinct policy title values from Qdrant.
    Requires Bearer token (same guard as /api/chat).

    from fastapi import APIRouter, Depends, HTTPException

    from backend.app.db.models import User
    from backend.app.services.auth import get_current_user
    from backend.app.services import rag

    router = APIRouter()


    @router.get("/sources")
    async def list_sources(
        current_user: User = Depends(get_current_user),
    ) -> dict:
        """
        Return sorted list of distinct payload.title values from the Qdrant policies collection.
        Uses facet API — O(1) aggregation (not a full scroll).
        Response: {"sources": ["Google Privacy Policy", "OpenAI Privacy Policy", ...]}
        """

            return {"sources": sources}
        except Exception:

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client → POST /api/chat | source_filter crosses here — user-controlled string enters backend |
| client → GET /api/sources | unauthenticated request must be rejected |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-09-01 | Information Disclosure | GET /api/sources | mitigate | Depends(get_current_user) guard required — policy titles are internal data; unauthenticated enumeration blocked |
| T-09-02 | Tampering | ChatRequest.source_filter → Qdrant MatchValue | accept | Pydantic validates type as str \| None; qdrant-client uses typed MatchValue(value=...) — no string interpolation into queries; injection is not possible via the filter parameter |
| T-09-03 | Tampering | source_filter as prompt injection vector | accept | source_filter scopes Qdrant retrieval only — it does not enter the LLM system prompt; no injection risk |
| T-09-04 | Spoofing | GET /api/sources with no token | mitigate | get_current_user raises HTTP 401 when no valid Bearer token present — same as /api/chat |
